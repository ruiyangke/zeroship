**Goal:** Add full WhatWG Fetch API (fetch, Request, Response, Headers) to the raw V8 runtime.

**Architecture:** One native Rust op (`__rawFetch`) bridges HTTP requests via reqwest to V8 promises through the existing async op channel. The full spec API surface (Request, Response, Headers, Body mixin) is implemented in JS loaded at isolate init. The tokio runtime Handle is stored in a V8 isolate slot so fetch callbacks can spawn async work.

**Tech Stack:** reqwest (HTTP client), serde_json (serialization), tokio (async runtime), v8 (JS engine)

```rust
//! Native fetch bridge — `__rawFetch(method, url, headersJson, body)` → Promise<String>.
//!
//! Returns a JSON string: `{"status":200,"statusText":"OK","headers":[["content-type","application/json"]],"body":"..."}`.
//! Errors resolve as: `{"error":"connection refused"}`.

use crate::event_loop::{OpResult, SharedState};

/// V8 callback for `__rawFetch(method, url, headersJson, body)`.
pub(crate) fn raw_fetch_callback(
    scope: &mut v8::PinScope,
    args: v8::FunctionCallbackArguments,
    mut rv: v8::ReturnValue,
) {
    let state: SharedState = scope
        .get_slot::<SharedState>()
        .expect("EventLoopState not in isolate slot")
        .clone();

    // Extract args
    let method = args.get(0).to_rust_string_lossy(scope);
    let url = args.get(1).to_rust_string_lossy(scope);
    let headers_json = args.get(2).to_rust_string_lossy(scope);
    let body = args.get(3).to_rust_string_lossy(scope);

    // Create promise
    let resolver = v8::PromiseResolver::new(scope).unwrap();
    let promise = resolver.get_promise(scope);

    // Register op
    let (op_id, op_tx, tokio_handle) = {
        let mut s = state.borrow_mut();
        let id = s.next_op_id;
        s.next_op_id += 1;
        s.pending_resolvers
            .insert(id, v8::Global::new(scope, resolver));
        (id, s.op_tx.clone(), s.tokio_handle.clone())
    };

    // Spawn HTTP request on tokio
    match tokio_handle {
        Some(handle) => {
            handle.spawn(async move {
                let result = do_fetch(method, url, headers_json, body).await;
                let _ = op_tx.send(OpResult {
                    id: op_id,
                    value: result,
                });
            });
        }
        None => {
            // No tokio runtime — do blocking fetch on a thread
            std::thread::spawn(move || {
                let rt = tokio::runtime::Builder::new_current_thread()
                    .enable_all()
                    .build()
                    .unwrap();
                let result = rt.block_on(do_fetch(method, url, headers_json, body));
                let _ = op_tx.send(OpResult {
                    id: op_id,
                    value: result,
                });
            });
        }
    }

    rv.set(promise.into());
}

/// Perform the HTTP request and return a JSON string with the response.
async fn do_fetch(method: String, url: String, headers_json: String, body: String) -> String {
    let client = reqwest::Client::builder()
        .timeout(std::time::Duration::from_secs(30))
        .build()
        .unwrap_or_else(|_| reqwest::Client::new());

    let req_method = match method.to_uppercase().as_str() {
        "GET" => reqwest::Method::GET,
        "POST" => reqwest::Method::POST,
        "PUT" => reqwest::Method::PUT,
        "DELETE" => reqwest::Method::DELETE,
        "PATCH" => reqwest::Method::PATCH,
        "HEAD" => reqwest::Method::HEAD,
        "OPTIONS" => reqwest::Method::OPTIONS,
        _ => reqwest::Method::GET,
    };

    let mut req = client.request(req_method, &url);

    // Parse headers from JSON: [["key","value"],...]
    if !headers_json.is_empty() && headers_json != "{}" && headers_json != "[]" {
        if let Ok(pairs) = serde_json::from_str::<Vec<(String, String)>>(&headers_json) {
            for (k, v) in pairs {
                req = req.header(&k, &v);
            }
        } else if let Ok(map) = serde_json::from_str::<serde_json::Map<String, serde_json::Value>>(&headers_json) {
            for (k, v) in map {
                if let Some(s) = v.as_str() {
                    req = req.header(&k, s);
                }
            }
        }
    }

    // Set body for non-GET requests
    if !body.is_empty() {
        req = req.body(body);
    }

    match req.send().await {
        Ok(resp) => {
            let status = resp.status().as_u16();
            let status_text = resp.status().canonical_reason().unwrap_or("").to_string();
            let headers: Vec<(String, String)> = resp
                .headers()
                .iter()
                .map(|(k, v)| {
                    (
                        k.to_string(),
                        v.to_str().unwrap_or("").to_string(),
                    )
                })
                .collect();
            let resp_url = resp.url().to_string();
            let redirected = resp_url != url;
            let body_text = resp.text().await.unwrap_or_default();

            serde_json::json!({
                "status": status,
                "statusText": status_text,
                "headers": headers,
                "body": body_text,
                "url": resp_url,
                "redirected": redirected,
            })
            .to_string()
        }
        Err(e) => {
            serde_json::json!({
                "error": e.to_string()
            })
            .to_string()
        }
    }
}
```

```javascript
// WhatWG Fetch API polyfill — built on __rawFetch native bridge.

(function(globalThis) {
"use strict";

// =========================================================================
// Headers
// =========================================================================

class Headers {
    constructor(init) {
        this._headers = [];
        if (init instanceof Headers) {
            init.forEach(function(value, name) { this.append(name, value); }, this);
        } else if (Array.isArray(init)) {
            for (var i = 0; i < init.length; i++) {
                if (init[i].length !== 2) throw new TypeError("Invalid header pair");
                this.append(init[i][0], init[i][1]);
            }
        } else if (init && typeof init === "object") {
            var keys = Object.keys(init);
            for (var j = 0; j < keys.length; j++) {
                this.append(keys[j], init[keys[j]]);
            }
        }
    }

    append(name, value) {
        name = String(name).toLowerCase();
        value = String(value);
        this._headers.push([name, value]);
    }

    delete(name) {
        name = String(name).toLowerCase();
        this._headers = this._headers.filter(function(h) { return h[0] !== name; });
    }

    get(name) {
        name = String(name).toLowerCase();
        var values = [];
        for (var i = 0; i < this._headers.length; i++) {
            if (this._headers[i][0] === name) values.push(this._headers[i][1]);
        }
        return values.length > 0 ? values.join(", ") : null;
    }

    has(name) {
        name = String(name).toLowerCase();
        for (var i = 0; i < this._headers.length; i++) {
            if (this._headers[i][0] === name) return true;
        }
        return false;
    }

    set(name, value) {
        name = String(name).toLowerCase();
        value = String(value);
        var found = false;
        this._headers = this._headers.filter(function(h) {
            if (h[0] === name) {
                if (!found) { found = true; h[1] = value; return true; }
                return false;
            }
            return true;
        });
        if (!found) this._headers.push([name, value]);
    }

    forEach(callback, thisArg) {
        for (var i = 0; i < this._headers.length; i++) {
            callback.call(thisArg, this._headers[i][1], this._headers[i][0], this);
        }
    }

    entries() { return this._headers[Symbol.iterator] ? this._headers.values() : this._headers; }
    keys() {
        var seen = {};
        var result = [];
        for (var i = 0; i < this._headers.length; i++) {
            if (!seen[this._headers[i][0]]) {
                seen[this._headers[i][0]] = true;
                result.push(this._headers[i][0]);
            }
        }
        return result;
    }
    values() {
        var self = this;
        return this.keys().map(function(k) { return self.get(k); });
    }

    // Serialize for native bridge
    _toArray() { return this._headers; }
}

Headers.prototype[Symbol.iterator] = function() { return this.entries(); };
```

```javascript
// =========================================================================
// Body mixin helpers
// =========================================================================

function consumeBody(obj) {
    if (obj._bodyUsed) throw new TypeError("Body already consumed");
    obj._bodyUsed = true;
    return obj._body;
}

function bodyMixin(proto) {
    Object.defineProperty(proto, "bodyUsed", {
        get: function() { return !!this._bodyUsed; }
    });
    proto.text = function() {
        return Promise.resolve(consumeBody(this));
    };
    proto.json = function() {
        return this.text().then(JSON.parse);
    };
    proto.arrayBuffer = function() {
        return this.text().then(function(text) {
            var encoder = typeof TextEncoder !== "undefined" ? new TextEncoder() : null;
            if (encoder) return encoder.encode(text).buffer;
            // Fallback: manual UTF-8 encode
            var buf = new ArrayBuffer(text.length);
            var view = new Uint8Array(buf);
            for (var i = 0; i < text.length; i++) view[i] = text.charCodeAt(i) & 0xff;
            return buf;
        });
    };
    proto.blob = function() {
        return this.text().then(function(text) {
            return { size: text.length, type: "", text: function() { return Promise.resolve(text); } };
        });
    };
    proto.formData = function() {
        return Promise.reject(new TypeError("formData() not supported"));
    };
}

// =========================================================================
// Request
// =========================================================================

class Request {
    constructor(input, init) {
        init = init || {};
        if (input instanceof Request) {
            this.url = input.url;
            this.method = input.method;
            this.headers = new Headers(input.headers);
            this._body = input._body || "";
            this.redirect = input.redirect;
            this.signal = input.signal;
        } else {
            this.url = String(input);
            this.method = (init.method || "GET").toUpperCase();
            this.headers = new Headers(init.headers);
            this._body = init.body || "";
            this.redirect = init.redirect || "follow";
            this.signal = init.signal || null;
        }
        this._bodyUsed = false;
        this.cache = init.cache || "default";
        this.credentials = init.credentials || "same-origin";
        this.mode = init.mode || "cors";
        this.referrer = init.referrer || "about:client";
    }

    clone() {
        return new Request(this);
    }
}

bodyMixin(Request.prototype);

// =========================================================================
// Response
// =========================================================================

class Response {
    constructor(body, init) {
        init = init || {};
        this._body = body != null ? String(body) : "";
        this._bodyUsed = false;
        this.status = init.status !== undefined ? init.status : 200;
        this.statusText = init.statusText || "";
        this.ok = this.status >= 200 && this.status < 300;
        this.headers = new Headers(init.headers);
        this.type = init.type || "basic";
        this.url = init.url || "";
        this.redirected = init.redirected || false;
    }

    clone() {
        if (this._bodyUsed) throw new TypeError("Cannot clone a consumed response");
        return new Response(this._body, {
            status: this.status,
            statusText: this.statusText,
            headers: new Headers(this.headers),
            url: this.url,
            redirected: this.redirected,
        });
    }

    static error() {
        var resp = new Response(null, { status: 0, statusText: "" });
        resp.type = "error";
        return resp;
    }

    static redirect(url, status) {
        if ([301, 302, 303, 307, 308].indexOf(status) === -1) {
            throw new RangeError("Invalid redirect status: " + status);
        }
        return new Response(null, {
            status: status,
            headers: { Location: url },
        });
    }

    static json(data, init) {
        init = init || {};
        var headers = new Headers(init.headers);
        if (!headers.has("content-type")) {
            headers.set("content-type", "application/json");
        }
        return new Response(JSON.stringify(data), {
            status: init.status || 200,
            statusText: init.statusText || "",
            headers: headers,
        });
    }
}

bodyMixin(Response.prototype);

// AbortController / AbortSignal
// =========================================================================

class AbortSignal {
    constructor() {
        this.aborted = false;
        this.reason = undefined;
        this._listeners = [];
    }
    addEventListener(type, listener) {
        if (type === "abort") this._listeners.push(listener);
    }
    removeEventListener(type, listener) {
        if (type === "abort") {
            this._listeners = this._listeners.filter(function(l) { return l !== listener; });
        }
    }
    throwIfAborted() {
        if (this.aborted) throw this.reason;
    }
    static abort(reason) {
        var signal = new AbortSignal();
        signal.aborted = true;
        signal.reason = reason || new DOMException("The operation was aborted.", "AbortError");
        return signal;
    }
    static timeout(ms) {
        var signal = new AbortSignal();
        setTimeout(function() {
            signal.aborted = true;
            signal.reason = new DOMException("The operation timed out.", "TimeoutError");
            for (var i = 0; i < signal._listeners.length; i++) signal._listeners[i]();
        }, ms);
        return signal;
    }
}

class AbortController {
    constructor() {
        this.signal = new AbortSignal();
    }
    abort(reason) {
        if (this.signal.aborted) return;
        this.signal.aborted = true;
        this.signal.reason = reason || new DOMException("The operation was aborted.", "AbortError");
        for (var i = 0; i < this.signal._listeners.length; i++) this.signal._listeners[i]();
    }
}

// DOMException polyfill (minimal)
if (typeof DOMException === "undefined") {
    globalThis.DOMException = function DOMException(message, name) {
        this.message = message;
        this.name = name || "Error";
    };
    globalThis.DOMException.prototype = Object.create(Error.prototype);
}

// =========================================================================
// fetch()
// =========================================================================

function fetch(input, init) {
    return new Promise(function(resolve, reject) {
        var request;
        try {
            request = input instanceof Request ? input : new Request(input, init);
        } catch (e) {
            return reject(e);
        }

        // Check abort
        if (request.signal && request.signal.aborted) {
            return reject(request.signal.reason);
        }

        // Serialize headers for native bridge
        var headerPairs = request.headers._toArray();

        // Call native bridge
        var rawPromise = __rawFetch(
            request.method,
            request.url,
            JSON.stringify(headerPairs),
            request._body || ""
        );

        // Wire abort signal
        var abortHandler;
        if (request.signal) {
            abortHandler = function() {
                reject(request.signal.reason);
            };
            request.signal.addEventListener("abort", abortHandler);
        }

        rawPromise.then(function(rawJson) {
            if (request.signal) request.signal.removeEventListener("abort", abortHandler);

            var raw;
            try { raw = JSON.parse(rawJson); } catch (e) {
                return reject(new TypeError("Invalid fetch response: " + e.message));
            }

            if (raw.error) {
                return reject(new TypeError("fetch failed: " + raw.error));
            }

            var responseHeaders = new Headers(raw.headers);
            var response = new Response(raw.body, {
                status: raw.status,
                statusText: raw.statusText,
                headers: responseHeaders,
                url: raw.url,
                redirected: raw.redirected,
            });

            resolve(response);
        }, function(err) {
            if (request.signal) request.signal.removeEventListener("abort", abortHandler);
            reject(new TypeError("fetch failed: " + err));
        });
    });
}

// =========================================================================
// Export to globalThis
// =========================================================================

globalThis.fetch = fetch;
globalThis.Headers = Headers;
globalThis.Request = Request;
globalThis.Response = Response;
globalThis.AbortController = AbortController;
globalThis.AbortSignal = AbortSignal;

})(globalThis);
```

### Task 8: Wire fetch into ConcurrentIsolate event loop

The per-request Isolate uses `op_tx`/`op_rx` directly. The ConcurrentIsolate uses `Event::OpCompleted` through its own channel. For the concurrent model, fetch ops must flow through the `event_tx` channel, not `op_tx`.

**Files:**
- Modify: `crates/isolate_v8/src/fetch.rs`
- Modify: `crates/isolate_v8/src/event_loop.rs`

**Step 1: Add an optional event_tx sender to EventLoopState**

In `event_loop.rs`, add:

```rust
pub(crate) struct EventLoopState {
    // ... existing fields ...
    /// Optional sender for the ConcurrentIsolate event channel.
    /// When set, async ops send Event::OpCompleted here instead of op_tx.
    pub(crate) concurrent_event_tx: Option<std::sync::mpsc::Sender<crate::concurrent::Event>>,
}
```

Initialize as `None` in `EventLoopState::new()`. Set it in `ConcurrentIsolate::new`:

```rust
el_state.borrow_mut().concurrent_event_tx = Some(event_tx.clone());
```

**Step 2: Update fetch.rs to use concurrent_event_tx when available**

In `raw_fetch_callback`, after the async work completes, check which channel to use:

```rust
let (op_id, op_tx, concurrent_tx, tokio_handle) = {
    let mut s = state.borrow_mut();
    let id = s.next_op_id;
    s.next_op_id += 1;
    s.pending_resolvers.insert(id, v8::Global::new(scope, resolver));
    (id, s.op_tx.clone(), s.concurrent_event_tx.clone(), s.tokio_handle.clone())
};

// In the spawned task:
match concurrent_tx {
    Some(tx) => {
        let _ = tx.send(crate::concurrent::Event::OpCompleted { op_id, value: result });
    }
    None => {
        let _ = op_tx.send(OpResult { id: op_id, value: result });
    }
}
```

**Step 3: Test with concurrent model**

Add test:

```rust
#[test]
fn concurrent_fetch_works() {
    init_v8();
    let js = r#"
        var __rpc = {
            test: async function() {
                var resp = await fetch("https://httpbin.org/get");
                var data = await resp.json();
                return data.url;
            }
        };
    "#;
    // Use ConcurrentIsolate
    let (event_tx, event_rx) = std::sync::mpsc::channel();
    let event_tx_clone = event_tx.clone();
    let handle = std::thread::spawn(move || {
        let mut isolate = ConcurrentIsolate::new(js, event_rx, event_tx_clone, None);
        isolate.run_until_idle();
    });
    let (reply_tx, reply_rx) = tokio::sync::oneshot::channel();
    event_tx.send(Event::NewRequest {
        id: 1,
        body: r#"{"jsonrpc":"2.0","method":"test","params":[],"id":1}"#.to_string(),
        reply: reply_tx,
    }).unwrap();
    drop(event_tx);
    let result = reply_rx.blocking_recv().unwrap().unwrap();
    assert!(result.json.contains("httpbin.org/get"));
    handle.join().unwrap();
}
```
